//circle method

#include<iostream>
using namespace std;

int main()
{
    int n1, n2, n;

    cout << "enter number of first signal samples:";
    cin >> n1;

    cout << "enter number of second signal samples:";
    cin >> n2;

    if(n1 <= 0 || n2 <= 0 || n1 > 100 || n2 > 100)
    {
        cout << "invalid sample size";
        return 0;
    }

    if(n1 > n2)
        n = n1;
    else
        n = n2;

    int x[100], y[100], h[100];

    // Zero padding
    for(int i = 0; i < n; i++)
    {
        x[i] = 0;
        h[i] = 0;
        y[i] = 0;
    }

    cout << "enter first signal:";
    for(int i = 0; i < n1; i++)
    {
        cin >> x[i];
    }

    cout << "enter second signal:";
    for(int i = 0; i < n2; i++)
    {
        cin >> h[i];
    }

    // Circular convolution
    for(int i = 0; i < n; i++)
    {
        y[i] = 0;

        for(int j = 0; j < n; j++)
        {
            y[i] += x[j] * h[(i-j+n)%n];
        }
    }

    cout << "circular convolution:";
    for(int i = 0; i < n; i++)
    {
        cout << y[i] << " ";
    }

    return 0;
}


//matrix method

#include<iostream>
using namespace std;

int main()
{
    int n1, n2, n;

    cout << "enter number of first signal samples:";
    cin >> n1;

    cout << "enter number of second signal samples:";
    cin >> n2;

    if(n1 <= 0 || n2 <= 0 || n1 > 100 || n2 > 100)
    {
        cout << "invalid sample size";
        return 0;
    }

    if(n1 > n2)
        n = n1;
    else
        n = n2;

    int x[100], y[100], h[100];
    int a[100][100];

    // Zero padding
    for(int i = 0; i < n; i++)
    {
        x[i] = 0;
        h[i] = 0;
        y[i] = 0;
    }

    cout << "enter first signal:";
    for(int i = 0; i < n1; i++)
        cin >> x[i];

    cout << "enter second signal:";
    for(int i = 0; i < n2; i++)
        cin >> h[i];

    // Matrix creation
    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < n; j++)
        {
            a[i][j] = h[(i-j+n)%n];
        }
    }

    // Matrix multiplication
    for(int i = 0; i < n; i++)
    {
        y[i] = 0;
        for(int j = 0; j < n; j++)
        {
            y[i] += x[j] * a[i][j];
        }
    }

    cout << "circular convolution:";
    for(int i = 0; i < n; i++)
        cout << y[i] << " ";

    return 0;
}


//deconvolution


#include <iostream>
using namespace std;

int main() {
    int i, j, x_size, y_size, h_size;
    int sum, x_o, y_o, h_o;

    cout << "Enter number of output samples: ";
    cin >> y_size;

    cout << "Enter number of impulses: ";
    cin >> h_size;

    if (y_size < h_size || h_size <= 0) {
        cout << "Invalid sample sizes!";
        return 0;
    }

    x_size = y_size - h_size + 1;

    int y[y_size], h[h_size], x[x_size];

    cout << "Enter output sequences: ";
    for (i = 0; i < y_size; i++) {
        cin >> y[i];
    }

    cout << "Enter impulse sequences: ";
    for (i = 0; i < h_size; i++) {
        cin >> h[i];
    }

    cout << "Enter origin point of y(n): ";
    cin >> y_o;

    cout << "Enter origin point of h(n): ";
    cin >> h_o;

    x_o = y_o - h_o;

    if (h[0] == 0) {
        cout << "Deconvolution is not possible with h[0] = 0.";
        return 0;
    }

    for (int i = 0; i < x_size; i++)
{
    sum = 0;

    for (int j = 0; j < i; j++)
    {
        if (i - j < h_size)
            sum += x[j] * h[i - j];
    }

    x[i] = (y[i] - sum) / h[0];
}

    cout << "\nInput sample values: ";
    for (i = 0; i < x_size; i++) {
        cout << x[i] << "  ";
    }

    cout << "\nOrigin point of input: " << x_o << endl;

    return 0;
}

//odd-even


#include <iostream>
using namespace std;

int main() {
    int n, origin;
    double x[100], even[100], odd[100];

    cout << "Enter number of samples: ";
    cin >> n;

    cout << "Enter signal samples: ";
    for (int i = 0; i < n; i++) {
        cin >> x[i];
    }

    cout << "Enter origin index (0 to " << n-1 << "): ";
    cin >> origin;

    cout << "Even component: ";
    for (int i = 0; i < n; i++) {
        int j = 2 * origin - i;
        double reversed = 0;

        if (j >= 0 && j < n)
            reversed = x[j];

        even[i] = (x[i] + reversed) / 2;
        cout << even[i] << " ";
    }

    cout << "\nOdd component: ";
    for (int i = 0; i < n; i++) {
        int j = 2 * origin - i;
        double reversed = 0;

        if (j >= 0 && j < n)
            reversed = x[j];

        odd[i] = (x[i] - reversed) / 2;
        cout << odd[i] << " ";
    }

    return 0;
}
